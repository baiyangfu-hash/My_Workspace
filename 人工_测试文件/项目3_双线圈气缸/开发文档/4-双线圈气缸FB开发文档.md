#### 3.2.2 双线圈控制逻辑

```scl
// 网络1：双线圈气缸控制逻辑
// 当i_气缸_S1闭合时，激活动点线圈，禁用原点线圈
IF i_气缸_S1 THEN
    q_气缸动点线圈 := TRUE;
    q_气缸原点线圈 := FALSE;
END_IF;

// 当i_气缸_S0闭合时，激活原点线圈，禁用动点线圈
IF i_气缸_S0 THEN
    q_气缸动点线圈 := FALSE;
    q_气缸原点线圈 := TRUE;
END_IF;

// 线圈互斥保护
// 确保两个线圈不会同时激活
IF q_气缸动点线圈 AND q_气缸原点线圈 THEN
    // 线圈冲突，禁用所有线圈
    q_气缸动点线圈 := FALSE;
    q_气缸原点线圈 := FALSE;
    q_原点异常 := TRUE;
    n_State := 3; // 切换到报警状态
END_IF;

// 控制信号冲突保护
IF i_气缸_S1 AND i_气缸_S0 THEN
    // 控制信号冲突，禁用所有线圈
    q_气缸动点线圈 := FALSE;
    q_气缸原点线圈 := FALSE;
    q_原点异常 := TRUE;
    n_State := 3; // 切换到报警状态
END_IF;
```

#### 3.2.3 传感器监控逻辑

```scl
// 网络2：气缸原点状态监控
// 正常原点到位延时定时器
T_0[0] := NOT q_气缸动点线圈 AND NOT i_屏蔽原点 AND i_原点传感器;
TONR(IN := T_0[0], 
     PT := 原点到位延时[0], 
     R := i_复位 OR q_气缸动点线圈, 
     Q => M_0[0], 
     ET => MW_0[0]);

// 屏蔽原点到位延时定时器
T_0[1] := NOT q_气缸动点线圈 AND i_屏蔽原点;
TONR(IN := T_0[1], 
     PT := 原点到位延时[1], 
     R := i_复位 OR q_气缸动点线圈, 
     Q => M_0[1], 
     ET => MW_0[1]);

// 网络3：气缸动点状态监控
// 正常动点到位延时定时器
T_0[3] := q_气缸动点线圈 AND NOT i_屏蔽动点 AND i_动点传感器;
TONR(IN := T_0[3], 
     PT := 动点到位延时[0], 
     R := i_复位 OR NOT q_气缸动点线圈, 
     Q => M_0[3], 
     ET => MW_0[3]);

// 屏蔽动点到位延时定时器
T_0[4] := q_气缸动点线圈 AND i_屏蔽动点;
TONR(IN := T_0[4], 
     PT := 动点到位延时[1], 
     R := i_复位 OR NOT q_气缸动点线圈, 
     Q => M_0[4], 
     ET => MW_0[4]);

// 到位状态互斥逻辑
// 原点到位状态
IF (M_0[0] OR M_0[1]) THEN
    q_气缸原点到位 := TRUE;
    q_气缸动点到位 := FALSE;
// 动点到位状态
ELSIF (M_0[3] OR M_0[4]) THEN
    q_气缸原点到位 := FALSE;
    q_气缸动点到位 := TRUE;
// 未知状态
ELSE
    q_气缸原点到位 := FALSE;
    q_气缸动点到位 := FALSE;
END_IF;

// 传感器故障检测
IF i_原点传感器 AND i_动点传感器 THEN
    // 传感器故障处理
    q_原点异常 := TRUE;
    q_动点异常 := TRUE;
    n_State := 3; // 切换到报警状态
END_IF;
```

#### 3.2.4 报警系统逻辑

```scl
// 网络4：原点报警
// 原点报警定时器
T_0[5] := NOT q_气缸动点线圈 AND NOT i_屏蔽原点 AND NOT i_原点传感器;
TONR(IN := T_0[5], 
     PT := 报警延时, 
     R := i_复位 OR NOT (NOT q_气缸动点线圈 AND NOT i_屏蔽原点 AND NOT i_原点传感器), 
     Q => M_0[8], 
     ET => MW_0[8]);

// 原点报警触发与自动恢复
IF M_0[8] THEN
    q_原点异常 := TRUE;
    n_State := 3; // 切换到报警状态
ELSIF NOT (NOT q_气缸动点线圈 AND NOT i_屏蔽原点 AND NOT i_原点传感器) THEN
    // 当报警条件消失时，自动恢复
    q_原点异常 := FALSE;
END_IF;

// 网络5：动点报警
// 动点报警定时器
T_0[6] := q_气缸动点线圈 AND NOT i_屏蔽动点 AND NOT i_动点传感器;
TONR(IN := T_0[6], 
     PT := 报警延时, 
     R := i_复位 OR NOT (q_气缸动点线圈 AND NOT i_屏蔽动点 AND NOT i_动点传感器), 
     Q => M_0[9], 
     ET => MW_0[9]);

// 动点报警触发与自动恢复
IF M_0[9] THEN
    q_动点异常 := TRUE;
    n_State := 3; // 切换到报警状态
ELSIF NOT (q_气缸动点线圈 AND NOT i_屏蔽动点 AND NOT i_动点传感器) THEN
    // 当报警条件消失时，自动恢复
    q_动点异常 := FALSE;
END_IF;

// 网络6：报警复位
// 当i_复位闭合时，复位异常状态
IF i_复位 THEN
    q_原点异常 := FALSE;
    q_动点异常 := FALSE;
    // 复位中间变量
    M_0[8] := FALSE;
    M_0[9] := FALSE;
END_IF;
```

#### 3.2.5 状态机管理逻辑

```scl
// 网络7：状态机管理
// 状态初始化
IF i_复位 THEN
    n_State := 0; // 复位到初始化状态
END_IF;

// 状态机逻辑
CASE n_State OF
    // 初始化状态
    0:
        // 检查初始状态
        IF q_气缸原点到位 THEN
            n_State := 1; // 切换到原点状态
        ELSIF q_气缸动点到位 THEN
            n_State := 2; // 切换到动点状态
        ELSE
            // 保持初始化状态
        END_IF;
    
    // 原点状态
    1:
        // 当气缸动点线圈激活时，切换到动点状态
        IF q_气缸动点线圈 THEN
            n_State := 2;
        END_IF;
    
    // 动点状态
    2:
        // 当气缸原点线圈激活且原点到位时，切换到原点状态
        IF q_气缸原点线圈 AND q_气缸原点到位 THEN
            n_State := 1;
        END_IF;
    
    // 报警状态
    3:
        // 当报警清除且回到正常状态时，切换到相应状态
        IF NOT q_动点异常 AND NOT q_原点异常 THEN
            IF q_气缸原点到位 THEN
                n_State := 1;
            ELSIF q_气缸动点到位 THEN
                n_State := 2;
            ELSE
                n_State := 0;
            END_IF;
        END_IF;
END_CASE;
```

## 7. 故障排除

### 7.1 常见故障

| 故障现象 | 可能原因 | 排除方法 |
|---------|---------|----------|
| 线圈不激活 | 控制信号未到达 | 检查控制信号线路 |
| 线圈不激活 | 线圈故障 | 检查线圈和继电器 |
| 传感器信号不稳定 | 传感器安装松动 | 重新固定传感器 |
| 传感器信号不稳定 | 电源干扰 | 添加滤波器，检查接地 |
| 报警持续触发 | 传感器位置不正确 | 调整传感器位置 |
| 报警持续触发 | 延时参数设置不当 | 调整延时参数 |
| 状态切换异常 | 状态机逻辑错误 | 检查状态机逻辑 |
| 状态切换异常 | 传感器信号异常 | 检查传感器状态 |

## 8. 开发注意事项

### 8.1 安全注意事项

1. **线圈互斥**：
   - 确保两个线圈不会同时激活，避免系统冲突
   - 实现多重互斥保护，提高系统可靠性

2. **电源安全**：
   - 线圈电源应独立供电，避免影响控制系统
   - 确保电源接地良好，避免电气干扰

3. **传感器安全**：
   - 传感器安装应牢固，避免松动导致信号异常
   - 定期检查传感器状态，确保正常工作

### 8.2 性能注意事项

1. **响应时间**：
   - 优化控制逻辑，减少响应时间
   - 合理设置定时器参数，避免过度延时

2. **资源占用**：
   - 优化变量使用，减少内存占用
   - 合理设计状态机，避免状态转换过于频繁

3. **可靠性**：
   - 增加错误检测和处理机制
   - 实现故障自动恢复功能

### 8.3 维护注意事项

1. **代码可维护性**：
   - 保持代码结构清晰，添加详细注释
   - 遵循命名规范，提高代码可读性
