### 1.1 Application (src/core/app.py)

```python
class Application:
    """应用主类"""
    
    属性:
        mode: str              # 运行模式 ("gui" / "cli")
        debug: bool            # 调试标志
        running: bool          # 运行状态
    
    方法:
        __init__(mode: str = "gui", debug: bool = False) -> None
        """初始化应用实例"""
        
        run() -> int
        """运行主循环，返回退出码 (0=正常, 1=异常)"""
        
        _run_gui() -> int
        """GUI模式: 创建QApplication和MainWindow"""
        
        shutdown() -> None
        """优雅关闭，保存状态"""
```

### 1.2 SettingsManager (src/core/settings.py)

```python
class SettingsManager:
    """设置管理器 - 单例模式"""
    
    类属性/方法:
        _settings_path: Path = None    # 配置文件路径（首次load时设置）
        _data: dict = {}              # 配置数据缓存
        
        load(path: Path) -> None
        """从JSON文件加载配置到内存"""
        
        save() -> None
        """将内存配置持久化到文件"""

        """读取配置项，不存在则返回默认值"""
        
        set(key: str, value: Any) -> None
        """写入配置项（仅内存，需save()持久化）"""
        
        add_recent_project(project_path: str, project_name: str) -> None
        """添加最近打开的项目（路径自动规范化为相对路径）"""
        
        get_recent_projects() -> List[Dict[str, str]]
        """获取最近项目列表（路径自动解析为绝对路径）"""
        
        _get_base_path() -> Optional[Path]
        """获取settings.json所在目录作为路径规范化的基准"""
```

### 1.3 EventBus (src/core/event_bus.py)

```python
class EventBus(QObject):
    """全局事件总线 - 单例模式"""
    
    信号:
        # 项目事件
        project_selected(str)           # 参数: 项目路径
        project_created(str)            # 参数: 新建项目路径
        project_opened(str)             # 参数: 打开的项目路径
        project_closed()                # 无参数
        
        # 文档事件
        document_open_request(str)      # 参数: 文档路径
        document_saved(str)              # 参数: 已保存的文档路径
        
        # 检查事件
        check_started()
        check_completed(object)           # 参数: CheckResult实例
        
        # 诊断事件
        diagnostic_started()
        diagnostic_completed(object)       # 参数: DiagnosticReport实例
    
    方法:
        @staticmethod
        instance() -> 'EventBus'
        """获取单例实例"""
```

### 1.4 ProjectService (src/services/project_service.py)

```python
class ProjectService:
    """项目管理服务"""
    
    @classmethod
    def create_project(
        cls,
        name: str,
        business_line: BusinessLine,
        template_id: str,
        manager: str = "",
        description: str = "",
        plc_brand=None,
        hmi_brand=None,
        custom_path: str = None
    ) -> Tuple[Optional[Project], Optional[str]]:
        """
        创建新项目
        
        Args:
            name: 项目名称
            business_line: 业务线枚举值
            template_id: 模板ID (M001/S001)
            manager: 项目经理
            description: 描述
            plc_brand: PLC品牌
            hmi_brand: HMI品牌
            custom_path: 自定义路径
            
        Returns:
            (Project实例或None, 错误信息或None)
        """
    
    @classmethod
    def open_project(cls, path: str) -> Optional[Project]:
        """打开已有项目，返回Project对象"""

    @classmethod
    def import_dj_project(cls, project_path: str) -> Tuple[Optional[Project], Optional[str]]:
        """导入现有DJ单机项目，自动识别资产并忽略 .trae/.plc-out"""
    
    @classmethod
    def close_project(cls) -> None:
        """关闭当前项目，清理状态"""
```

### 1.5 DiagnosticService (src/services/diagnostic_service.py)

```python
class DiagnosticService:
    """诊断服务"""
    
    @classmethod
    def run_diagnosis(cls, project_path: str) -> DiagnosticReport:
        """
        执行七维度深度诊断
        
        维度与权重:
        - 结构完整性 (15%): 目录/文件齐全性
        - 命名规范率 (15%): 符合命名规范的占比
        - 文档覆盖率 (15%): 必需文档存在情况
        - 代码质量分 (20%): 语法/注释综合评分
        - 配置合规性 (10%): .plc.json正确性
        - LSP兼容性 (15%): Siemens LSP兼容度
        - 测试覆盖度 (10%): 测试用例匹配度
        
        Returns:
            DiagnosticReport 实例
        """
    
    @classmethod
    def run_lsp_check(cls, project_path: str) -> LSPCheckResult:
        """执行LSP兼容性专项检查"""
```

## 2. settings.json 配置项说明

| 键名 | 类型 | 默认值 | 说明 |
|------|------|--------|------|
| `recent_projects` | List | `[]` | 最近打开的项目列表，每项含path(相对)和name |
| `max_recent_projects` | int | `10` | 最大保留数量 |
| `theme` | str | `"material_dark"` | UI主题名称 |
| `last_opened_project` | str/null | 上次打开的项目路径 |
| `window_geometry` | dict | `{width:1280, height:800}` | 窗口大小 |
| `default_project_root` | str | `"./Projects"` | 新建项目默认根目录 |
