class VersionExtractor:
    """版本号提取器 - 从不同文件格式中提取版本信息"""

    # 正则：匹配 "(* Vx.y.z *)" 格式（.scl文件头）
    RE_SCL_VERSION = re.compile(
        r'\(\*\s*V(\d+)\.(\d+)\.(\d+)\s*\*\)'
    )
    # 正则：匹配文件名中的 "Vx.y.z" 格式（PRD文档）
    RE_FILENAME_VERSION = re.compile(
        r'[Vv](\d+)\.(\d+)\.(\d+)'
    )
    # 正则：匹配 Markdown 内容中的版本行
    RE_MD_VERSION_LINE = re.compile(
        r'>\s*(?:版本|Version)\s*[：:]\s*V(\d+)\.(\d+)\.(\d+)'
    )

    @classmethod
    def extract_scl_version(cls, file_path: str) -> Optional[str]:
        """
        从 .scl 文件头提取版本号
        扫描文件前 10 行，匹配 "(* Vx.y.z *)" 模式
        Returns: "V7.1.1" 或 None
        """

    @classmethod
    def extract_prd_version(cls, file_path: str) -> Optional[str]:
        """
        从 PRD 文档提取版本号
        策略:
        1. 先尝试从文件名提取 (IFC-FB1003-PickPlace-V7.0.0.md → V7.0.0)
        2. 再尝试从文件内容前 20 行匹配版本声明行
        Returns: "V7.0.0" 或 None
        """

    @classmethod
    def extract_prd_version_from_filename(cls, file_name: str) -> Optional[str]:
        """
        从文件名提取版本号
        "接口文档_IFC-FB1003-PickPlace-V7.0.0.md" → "V7.0.0"
        """

    @classmethod
    def parse_version_tuple(cls, version_str: str) -> Tuple[int, int, int]:
        """
        将版本字符串解析为元组 ("V7.1.1" → (7, 1, 1))
        用于版本号比较
        """

    @classmethod
    def compare_versions(cls, v1: str, v2: str) -> int:
        """
        比较两个版本号
        Returns: -1(v1<v2), 0(v1==v2), 1(v1>v2)
        """
```

@dataclass
class VarRecord:
    """单条变量记录"""
    name: str                    # 变量名 (如 "i_bReset")
    type_: str                   # 数据类型 (如 "BOOL", "ARRAY[0..9] OF INT")
    default: str                 # 默认值 (如 "FALSE", "5000", "")
    comment: str                 # 行注释 (如 "全局复位按钮")

@dataclass
class StructGroup:
    """一个 STRUCT 结构体组"""
    name: str                    # 结构体名 (如 "stGlobal")
    description: str             # 块注释描述 (多行)
    header_comment: str          # 头部注释 (接口统计等)
    variables: List[VarRecord]   # 变量列表
    input_count: int             # 输入变量数 (i_前缀)
    output_count: int            # 输出变量数 (o_/q_前缀)

class DbParser:
    """GlobalVars.db 解析器"""

    RE_DATA_BLOCK = re.compile(r'DATA_BLOCK\s+(\w+)')
    RE_STRUCT_START = re.compile(r'(\w+)\s*:\s*STRUCT')
    RE_VAR_LINE = re.compile(
        r'^\s*(\w+)\s*:\s*(.+?);\s*(?://\s*(.*))?$'
    )
    RE_TYPE_WITH_DEFAULT = re.compile(
        r'(.+?)\s*:=\s*(.+)$'
    )

    @classmethod
    def parse_db_file(cls, file_path: str) -> List[StructGroup]:
        """
        解析 GlobalVars.db，提取所有 STRUCT 定义
        流程:
        1. 定位 DATA_BLOCK ... VAR 区块
        2. 按 stXxx : STRUCT ... END_STRUCT; 切分结构体
        3. 对每个 STRUCT 内的变量行逐行解析
        4. 提取变量名、类型、默认值、注释
        Returns: 按文件顺序排列的 StructGroup 列表
        """

    @classmethod
    def _parse_var_line(cls, line: str) -> Optional[VarRecord]:
        """解析单行变量声明 → VarRecord"""

    @classmethod
    def _extract_collect_comment(cls, lines: List[str], start: int) -> str:
        """向前收集连续的 // 注释块作为结构体描述"""

    @classmethod
    def _compute_io_counts(cls, variables: List[VarRecord]) -> Tuple[int, int]:
        """按命名前缀统计输入/输出数量"""

    """全量同步报告生成器 - L2 按需层"""

    @classmethod
    def generate_full_report(cls, project_path: str) -> FullSyncReport:
        """
        生成 L0+L1+L2 全量同步报告
        流程:
        1. VersionChecker.check_project() → 版本差距矩阵 (§1)
        2. 扫描所有 DSN 文档 → DSN 状态表 (§2)
        3. 扫描所有 UM 文档 → UM 状态表 (§3)
        4. PM_SESSION change_log 聚合 → 变更台帐 (§5)
        5. 综合分析 → 行动建议优先级排序 (§6)
        """

    @classmethod
    def render_markdown(cls, report: FullSyncReport) -> str:
        """
        将 FullSyncReport 渲染为完整 Markdown 报告
        六段式结构 (§1~§6)
        """

    @classmethod
    def _check_documents(cls, project_path: str, doc_type: str,
                          version_gaps: List[FbVersionGap]) -> List[DocStatus]:
        """
        检查指定类型文档的覆盖度和版本对齐
        doc_type: "DSN" | "UM"
        """

    @classmethod
    def _aggregate_ledger(cls, session_path: str) -> List[ChangeLogEntry]:
        """
        从 PM_SESSION 聚合 change_log 为变更台帐条目
        """

    @classmethod
    def _generate_action_items(cls, report: FullSyncReport) -> List[str]:
        """
        根据报告中的差距自动生成优先级排序的行动建议
        优先级: P0(代码无文档) > P1(版本滞后>1) > P2(文件名无版本)
        """
```
